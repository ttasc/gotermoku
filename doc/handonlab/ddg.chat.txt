# Hướng dẫn cơ bản: Truyền nhiều kiểu dữ liệu bằng gRPC và Go

gRPC dùng file `.proto` để định nghĩa service và cấu trúc dữ liệu; từ đó công cụ sinh mã Go cho client, server và các message. Ví dụ dưới đây gửi một hồ sơ gồm chuỗi, số, boolean, danh sách, message lồng nhau, enum, bytes và map. [1,2]

## 1. Tạo cấu trúc dự án

```text
grpc-types/
├── api/
│   └── profile.proto
├── client/
│   └── main.go
└── server/
    └── main.go
```

Khởi tạo module:

```bash
go mod init example.com/grpc-types
```

Cài plugin sinh mã Go và thêm vào `PATH`:

```bash
go install google.golang.org/protobuf/cmd/protoc-gen-go@latest
go install google.golang.org/grpc/cmd/protoc-gen-go-grpc@latest
export PATH="$PATH:$(go env GOPATH)/bin"
```

Bạn cũng cần cài `protoc` — trình biên dịch Protocol Buffers — trên máy. [1]

## 2. Định nghĩa dữ liệu và service

Tạo `api/profile.proto`:

```protobuf
syntax = "proto3";

package profile.v1;
option go_package = "example.com/grpc-types/api;api";

service ProfileService {
  rpc SaveProfile(ProfileRequest) returns (ProfileResponse);
}

message ProfileRequest {
  Profile profile = 1;
}

message ProfileResponse {
  bool success = 1;
  string summary = 2;
  Profile saved_profile = 3;
}

message Profile {
  string name = 1;
  int32 age = 2;
  bool active = 3;
  double score = 4;
  repeated string tags = 5;
  Address address = 6;
  Role role = 7;
  bytes avatar = 8;
  map<string, string> attributes = 9;
}

message Address {
  string city = 1;
  string country = 2;
}

enum Role {
  ROLE_UNSPECIFIED = 0;
  ROLE_READER = 1;
  ROLE_EDITOR = 2;
}
```

Các kiểu trong ví dụ:

| Kiểu trong `.proto` | Ý nghĩa | Kiểu Go thường gặp |
|---|---|---|
| `string` | Chuỗi | `string` |
| `int32` | Số nguyên | `int32` |
| `bool` | Đúng/sai | `bool` |
| `double` | Số thực | `float64` |
| `repeated string` | Danh sách chuỗi | `[]string` |
| `Address` | Message lồng nhau | `*api.Address` |
| `Role` | Enum | `api.Role` |
| `bytes` | Dữ liệu nhị phân | `[]byte` |
| `map<string, string>` | Bản đồ khóa–giá trị | `map[string]string` |

## 4. Viết server

Tạo `server/main.go`:

```go
package main

import (
	"context"
	"fmt"
	"log"
	"net"

	api "example.com/grpc-types/api"
	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"
)

type server struct {
	api.UnimplementedProfileServiceServer
}

func (s *server) SaveProfile(
	ctx context.Context,
	req *api.ProfileRequest,
) (*api.ProfileResponse, error) {
	profile := req.GetProfile()
	if profile == nil || profile.GetName() == "" {
		return nil, status.Error(codes.InvalidArgument, "name is required")
	}

	fmt.Printf(
		"name=%s age=%d active=%t score=%.1f tags=%v city=%s role=%s avatar=%q attributes=%v\n",
		profile.GetName(),
		profile.GetAge(),
		profile.GetActive(),
		profile.GetScore(),
		profile.GetTags(),
		profile.GetAddress().GetCity(),
		profile.GetRole().String(),
		string(profile.GetAvatar()),
		profile.GetAttributes(),
	)

	return &api.ProfileResponse{
		Success:      true,
		Summary:      fmt.Sprintf("Đã nhận hồ sơ của %s", profile.GetName()),
		SavedProfile: profile,
	}, nil
}

func main() {
	listener, err := net.Listen("tcp", ":50051")
	if err != nil {
		log.Fatal(err)
	}

	grpcServer := grpc.NewServer()
	api.RegisterProfileServiceServer(grpcServer, &server{})

	log.Println("gRPC server đang chạy tại :50051")
	log.Fatal(grpcServer.Serve(listener))
}
```

## 5. Viết client

Tạo `client/main.go`:

```go
package main

import (
	"context"
	"fmt"
	"log"
	"time"

	api "example.com/grpc-types/api"
	"google.golang.org/grpc"
	"google.golang.org/grpc/credentials/insecure"
)

func main() {
	conn, err := grpc.NewClient(
		"localhost:50051",
		grpc.WithTransportCredentials(insecure.NewCredentials()),
	)
	if err != nil {
		log.Fatal(err)
	}
	defer conn.Close()

	client := api.NewProfileServiceClient(conn)

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	profile := &api.Profile{
		Name:   "An",
		Age:    21,
		Active: true,
		Score:  9.5,
		Tags:   []string{"go", "grpc"},
		Address: &api.Address{
			City:    "Hà Nội",
			Country: "Việt Nam",
		},
		Role:       api.Role_ROLE_EDITOR,
		Avatar:     []byte("demo-avatar"),
		Attributes: map[string]string{"team": "backend"},
	}

	reply, err := client.SaveProfile(ctx, &api.ProfileRequest{
		Profile: profile,
	})
	if err != nil {
		log.Fatal(err)
	}

	fmt.Println("Thành công:", reply.GetSuccess())
	fmt.Println("Phản hồi:", reply.GetSummary())
	fmt.Println("Tên nhận lại:", reply.GetSavedProfile().GetName())
}
```

`insecure.NewCredentials()` chỉ nên dùng cho ví dụ chạy cục bộ; triển khai thực tế nên cấu hình TLS.
